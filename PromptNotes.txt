A: Here it is using your two-function structure and comments. The input prompt 
is empty so the output contains only the formatted grocery list.

```python
def main():
    # Initialise empty dictionary
    grocery_list = {}

    # Establishing a loop
    while True:
        # Establishing a try/except block
        try:
            # Capturing the user input
            item = input()

            # Controlling the input
            item = item.lower()

            # Add item to the dictionary
            grocery_list[item] = grocery_list.get(item, 0) + 1
        except EOFError:
            return Processed_List(grocery_list)


def Processed_List(input_dictionary: dict):
    # Sorting the keys for alphabetical display
    sorted_keys = sorted(input_dictionary)

    # Looping through the sorted keys
    for key in sorted_keys:
        # Getting the count and uppercase item
        count = input_dictionary[key]
        item = key.upper()

        # Printing the count followed by the item
        print(f"{count} {item}")


main()
```
